/* -------------------------------- Snippet -------------------------------- */

export function openSnippet(id, pre = {}) {
  const S = store.S;
  const x = S.snippets.find((s) => s.id === id) || { q: pre.q || "", a: pre.a || "", enabled: true };
  modal(
    id ? "Edit snippet" : "New snippet",
    `<label class="fl" for="sq">Customer question</label>
<input class="in" id="sq" value="${esc(x.q)}" maxlength="300" placeholder="For example: Do you offer free setup?">
<label class="fl" for="sa">Answer</label>
<textarea class="in" id="sa" rows="8" maxlength="2000" placeholder="Write the facts the answer needs. The bot rewords it in its usual reply format.">${esc(x.a)}</textarea>
<p class="meta"><span id="sc">${x.a.length}</span>/2000 characters. English or Vietnamese both work.</p>
<label class="chk"><input type="checkbox" id="se" ${x.enabled ? "checked" : ""}> The bot uses this snippet</label>
${
  S.settings.snippetMode === "exact"
    ? `<p class="meta">All snippets are sent exactly as written (see "How the bot uses snippets" above the list).</p>`
    : `<label class="chk"><input type="checkbox" id="sx" ${x.exact ? "checked" : ""}> Always send this answer exactly as written <span class="muted">(for approved wording, like a policy)</span></label>`
}`,
    `<button class="btn" value="close">Cancel</button><button class="btn pri" type="button" data-act="snipSave" data-id="${esc(id || "")}">Save snippet</button>`,
    {
      onOpen(d) {
        $("#sa", d).oninput = (e) => ($("#sc", d).textContent = e.target.value.length);
        (id || pre.q ? $("#sa", d) : $("#sq", d)).focus();
      },
    }
  );
}

A.snipSave = (el) =>
  busy(el, async () => {
    const S = store.S;
    const prev = S.snippets.find((x) => x.id === el.dataset.id);
    // No tick box on screen (all snippets are sent as written, or the quick form): keep what the snippet had
    const exact = $("#sx") ? $("#sx").checked : !!prev?.exact;
    const item = { id: el.dataset.id || undefined, q: $("#sq").value, a: $("#sa").value, enabled: $("#se").checked, exact, src: UI.fixLog ? "gap" : undefined };
    const r = await api("/api/snippets", { op: "save", item, fixLog: UI.fixLog || undefined });
    S.snippets = r.snippets;
    const fixed = UI.fixLog;
    if (fixed) {
      const log = C.log[fixed.month];
      if (log) for (const it of log) if (fixed.ids.includes(it.id)) it.st = "fixed";
      if (log && fixed.month === S.month) {
        S.pending = log.filter((x) => x.gap && !x.st).length;
        window.dispatchEvent(new Event("tb:side"));
      }
    }
    const retry = UI.retry;
    UI.fixLog = null;
    UI.retry = null;
    closeModal();
    refresh();
    toast("Snippet saved. The bot uses it from the next message.", retry ? { action: "Test this question", onAction: () => pvAsk(retry) } : {});
  });

/* ---------------------------- Sources the bot used ---------------------------- */

export function srcInfo(s) {
  const S = store.S;
  const sn = S.snippets.find((x) => x.id === s);
  if (sn) return { kind: "snippet", id: sn.id, title: sn.q };
  const clean = String(s).replace(/[#?].*$/, "").replace(/\/$/, "");
  const a = S.kb.articles.find((x) => x.url.replace(/\/$/, "") === clean || (x.src === "help" && clean.endsWith(x.id.replace(/\/$/, ""))));
  if (a) return { kind: "article", id: a.id, url: a.url, title: a.title, off: a.off };
  if (/^https?:\/\//.test(s)) return { kind: "link", url: s, title: s.replace(/^https?:\/\//, "") };
  return null;
}

export function srcLinks(sources) {
  const list = (sources || []).map(srcInfo).filter(Boolean);
  if (!list.length) return "none";
  return list
    .map((s) =>
      s.kind === "snippet"
        ? `<button type="button" class="lk" data-act="snipEdit" data-id="${esc(s.id)}">Snippet: ${esc(s.title)}</button>`
        : `<a href="${esc(s.url)}" target="_blank" rel="noopener">${esc(s.title)}</a>`
    )
    .join(", ");
}

// Snippets closest to the question: sources the bot used first, then snippets sharing the most words
function related(q, sources = []) {
  const S = store.S;
  const used = sources.map(srcInfo).filter((x) => x?.kind === "snippet").map((x) => x.id);
  const toks = new Set(words(q).split(/[^a-z0-9]+/).filter((w) => w.length > 2 && !STOP.has(w)));
  const score = (s) => {
    const w = new Set(words(`${s.q} ${s.a}`).split(/[^a-z0-9]+/));
    let k = 0;
    for (const t of toks) if (w.has(t)) k++;
    return k;
  };
  return S.snippets
    .map((s) => ({ s, k: used.includes(s.id) ? 100 : score(s) }))
    .filter((x) => x.k > 0)
    .sort((a, b) => b.k - a.k)
    .slice(0, 6)
    .map((x) => ({ ...x.s, used: used.includes(x.s.id) }));
}

/* --------------------------- Improve this answer --------------------------- */

/** q: customer question; a: the bot's answer; sources: what the bot used; fix: {month, ids} to mark as handled on save */
export function openImprove({ q = "", a = "", sources = [], fix = null }) {
  UI.fixLog = fix && fix.ids?.length ? fix : null;
  UI.retry = q;
  const rel = related(q, sources);
  const arts = sources.map(srcInfo).filter((x) => x && x.kind !== "snippet");
  modal(
    "Improve this answer",
    `${q || a ? `<div class="ctx">${q ? `<b>Customer asked</b><p>${esc(q)}</p>` : ""}${a ? `<b>Bot answered</b><p>${esc(a)}</p>` : ""}</div>` : ""}
<h3>Create a new snippet</h3>
<p class="hint">Next time a customer asks something similar, the bot answers from this snippet.</p>
<label class="fl" for="sq">Question</label>
<input class="in" id="sq" value="${esc(q)}" maxlength="300">
<label class="fl" for="sa">Answer</label>
<textarea class="in" id="sa" rows="6" maxlength="2000" placeholder="Write the facts the answer needs."></textarea>
<label class="chk"><input type="checkbox" id="se" checked> Turn on now</label>
<p style="margin:12px 0 0"><button type="button" class="btn pri" data-act="snipSave" data-id="">Save snippet</button></p>
<h3>Or update related content</h3>
<div class="rows">${
      rel.length || arts.length
        ? rel
            .map(
              (s) => `<div class="row"><div class="grow"><p class="t">${esc(s.q)}${s.used ? ` <span class="chip c-handoff">Used by the bot</span>` : ""}</p><p class="sub clamp">${esc(s.a)}</p></div>
<div class="acts"><button type="button" class="btn sm" data-act="snipEdit" data-id="${esc(s.id)}">Edit</button></div></div>`
            )
            .join("") +
          arts
            .map(
              (x) => `<div class="row"><div class="grow"><p class="t">${esc(x.title)} <span class="chip">Used by the bot</span></p><p class="sub">An article the bot read. Edit it on the help center or website, then click Sync now in Knowledge.</p></div>
<div class="acts"><a class="btn sm" href="${esc(x.url)}" target="_blank" rel="noopener">Open article</a></div></div>`
            )
            .join("")
        : `<div class="empty">No snippets are close to this question.</div>`
    }</div>`,
    `<button class="btn" value="close">Close</button>`,
    { wide: true, onOpen: (d) => $("#sa", d).focus() }
  );
}

/* --------------------------------- Charts --------------------------------- */

export const fig = (value, label, sub = "") => `<div class="fig"><b>${value}</b><span>${label}</span>${sub ? `<small>${sub}</small>` : ""}</div>`;

function niceTicks(max) {
  const steps = [1, 2, 5, 10, 20, 25, 50, 100, 200, 250, 500, 1000, 2000, 5000];
  const step = steps.find((s) => max / s <= 4) || Math.ceil(max / 4);
  const top = Math.max(step, Math.ceil(max / step) * step);
  const out = [];
  for (let v = 0; v <= top; v += step) out.push(v);
  return out;
}

/** One bar per day, one colour, tooltip on hover or focus */
export function dailyChart(days, value, tipText, label) {
  const max = Math.max(1, ...days.map(value));
  const ticks = niceTicks(max);
  const top = ticks.at(-1);
  return `<div class="chart" role="img" aria-label="${esc(label)}">
<div class="plot">${ticks.map((t) => `<div class="gl${t === 0 ? " base" : ""}" style="bottom:${(t / top) * 100}%"><span>${n(t)}</span></div>`).join("")}
<div class="cols">${days
    .map((d) => `<div class="col" tabindex="0" data-tip="${esc(tipText(d))}"><i style="height:${(value(d) / top) * 100}%"></i></div>`)
    .join("")}</div></div>
<div class="xax" aria-hidden="true">${days.map((d) => `<span>${d.d === 1 || d.d % 5 === 0 ? d.d : ""}</span>`).join("")}</div></div>`;
}

export function funnel(steps) {
  const base = steps[0].v || 0;
  return `<div class="fun">${steps
    .map((s) => {
      const w = base ? (s.v / base) * 100 : 0;
      return `<div class="fs"><span>${esc(s.label)}</span><div class="fb"><i style="width:${w}%"></i></div><div class="fv">${n(s.v)}<small>${base ? Math.round(w) + "%" : ""}</small></div></div>`;
    })
    .join("")}</div>`;
}

// Tooltip for chart bars (hover or Tab)
function showTip(el) {
  const tip = $("#tip");
  const r = el.getBoundingClientRect();
  tip.textContent = el.dataset.tip;
  tip.style.left = `${Math.min(window.innerWidth - 130, Math.max(130, r.left + r.width / 2))}px`;
  tip.style.top = `${r.top + Math.max(0, r.height - (el.firstElementChild?.offsetHeight || 0))}px`;
  tip.classList.add("on");
}
const hideTip = () => $("#tip").classList.remove("on");
document.addEventListener("mouseover", (e) => {
  const c = e.target.closest?.("[data-tip]");
  c ? showTip(c) : hideTip();
});
document.addEventListener("focusin", (e) => (e.target.matches?.("[data-tip]") ? showTip(e.target) : hideTip()));
document.addEventListener("scroll", hideTip, true);
