// Admin entry point: load data, route by #page, sidebar, event delegation.
import { $, $$, A, R, UI, api, esc, fail, nav, shiftMonth, store, toast } from "./core.js";
import { initPreview, overlayState, pvClose } from "./preview.js";
import "./components.js";
import "./pages/overview.js";
import "./pages/content.js";
import "./pages/guidance.js";
import "./pages/tags.js";
import "./pages/escalation.js";
import "./pages/followup.js";
import "./pages/performance.js";
import "./pages/conversations.js";
import "./pages/suggestions.js";
import "./pages/settings.js";

const TRAIN = new Set(["content", "guidance", "tags", "escalation", "followup"]);
const MODE_NOTE = { off: "The bot is off", offline: "Replies when the team is offline", always: "Replies to every message" };
let current = "overview";
let currentArg = null;

function renderSide() {
  const S = store.S;
  const s = S.settings;
  $("#botName").textContent = s.botName;
  document.title = `${s.botName} Admin`;
  $(".top-name").textContent = s.botName;
  $$(".mark, .fab .av").forEach((el) => (el.textContent = (s.botName || "T").trim().charAt(0).toUpperCase()));
  $("#modeSel").value = s.mode;
  $("#modeDot").className = `dot${s.mode === "off" ? " off" : ""}`;
  $("#modeNote").textContent = MODE_NOTE[s.mode];
  const p = $("#nPending");
  p.textContent = S.pending > 99 ? "99+" : S.pending;
  p.hidden = !S.pending;
  $("#who").textContent = S.who;
}

function route() {
  const [name, arg] = (location.hash.slice(1) || "overview").split("?");
  current = R[name] ? name : "overview";
  currentArg = arg || null;
  nav.gen++;
  $$("#nav a").forEach((a) => (a.getAttribute("href") === `#${current}` ? a.setAttribute("aria-current", "page") : a.removeAttribute("aria-current")));
  $("#app").classList.toggle("dock", TRAIN.has(current));
  $("#side").classList.remove("open");
  overlayState();
  window.scrollTo(0, 0);
  Promise.resolve(R[current](currentArg)).catch(fail);
  $("#main").focus({ preventScroll: true });
}

// Click on an element with data-act="name" → run A[name]
document.addEventListener("click", (e) => {
  const el = e.target.closest("[data-act]");
  if (!el || el.disabled) return;
  const fn = A[el.dataset.act];
  if (!fn) return;
  e.preventDefault();
  Promise.resolve(fn(el, e)).catch(fail);
});

A.month = (el) => {
  UI.month = shiftMonth(UI.month || store.S.month, +el.dataset.d);
  nav.gen++;
  R[current](currentArg);
};
A.sideOpen = () => {
  $("#side").classList.add("open");
  overlayState();
};
A.closeOverlays = () => {
  $("#side").classList.remove("open");
  pvClose();
};
document.addEventListener("keydown", (e) => {
  if (e.key === "Escape" && !$("#dlg").open) A.closeOverlays();
});

window.addEventListener("hashchange", route);
window.addEventListener("tb:refresh", () => {
  nav.gen++;
  Promise.resolve(R[current](currentArg)).catch(fail);
});
window.addEventListener("tb:side", renderSide);
window.addEventListener("resize", overlayState);

$("#modeSel").addEventListener("change", async (e) => {
  const S = store.S;
  const before = S.settings.mode;
  try {
    const r = await api("/api/settings", { settings: { mode: e.target.value } });
    Object.assign(S, r);
    toast(`Bot mode: ${$("#modeSel").selectedOptions[0].textContent}`);
    if (current === "overview" || current === "settings") R[current]();
  } catch (err) {
    S.settings.mode = before;
    fail(err);
  }
  renderSide();
});

async function boot() {
  try {
    store.S = await api("/api/state");
  } catch (e) {
    $("#main").innerHTML = `<div class="wrap"><header class="ph"><div><h1>Couldn't load the admin</h1><p>${esc(e.message)}</p></div></header><p><a class="btn" href="/admin">Reload</a></p></div>`;
    return;
  }
  UI.month = store.S.month;
  renderSide();
  initPreview();
  route();
}
boot();
