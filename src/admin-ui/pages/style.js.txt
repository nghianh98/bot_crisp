// Writing style: a tool only. The team writes the rules, the introduction and the example replies; nothing about tone is fixed in code.
import { $, $$, A, R, ago, api, busy, closeModal, confirmBox, esc, header, matches, modal, n, plural, refresh, setMain, store, sw, toast } from "../core.js";
import { colName, guessColumns, parseDelimited, readSheetFile, toItems } from "../sheet.js";

let q = "";
// Import dialog: sheets read from the file, the chosen sheet, column choices and the rows (with tick state)
const imp = { sheets: [], si: 0, cols: null, items: [] };

R.style = () => {
  const S = store.S;
  const st = S.style;
  const M = S.meta;
  const on = st.examples.filter((e) => e.enabled).length;
  const list = st.examples.filter((e) => matches(q, e.group, e.situation, e.text));
  const groups = [...new Set(list.map((e) => e.group || ""))];
  const row = (e) => `<div class="row">${sw(e.enabled, "exToggle", `data-id="${esc(e.id)}"`, "The bot learns from this reply")}
<div class="grow"><button type="button" class="lk t" data-act="exEdit" data-id="${esc(e.id)}">${esc(e.situation || e.text.split("\n")[0].slice(0, 90))}</button><p class="sub clamp">${esc(e.text)}</p>
<p class="meta">Edited ${ago(e.upd)}${e.by ? ` by ${esc(e.by)}` : ""}</p></div>
<div class="acts"><button type="button" class="btn ghost sm danger" data-act="exDel" data-id="${esc(e.id)}">Delete</button></div></div>`;
  const isDefault = st.rules == null;
  setMain(`${header("Writing style", "How the bot writes, in your own words: the rules for every reply, how it introduces itself, and real replies from your team.")}
<section class="sec"><div class="sec-h"><h2>Writing rules</h2>${isDefault ? `<span class="chip">Default rules</span>` : ""}</div>
<p class="hint">Write how every reply should look, one rule per line: length, how to open and close, whether to sign off with a name, how to show steps and links. You can add an example reply at the end. Snippets are still sent as written. The chat can't show bold or bullet symbols, so the bot always writes plain text.</p>
<textarea class="in" id="stRules" rows="16" maxlength="${M.maxRulesChars}" spellcheck="false" placeholder="Leave empty to give the bot no writing rules">${esc(isDefault ? M.defaultRules : st.rules)}</textarea>
<div class="toolbar"><button type="button" class="btn pri" data-act="stRulesSave">Save rules</button>${isDefault ? "" : `<button type="button" class="btn ghost" data-act="stRulesReset">Restore default rules</button>`}<span class="meta" id="stRulesN" style="margin:0"></span></div></section>
<section class="sec"><h2>How the bot introduces itself</h2><p class="hint">Used once, in the bot's first reply of a conversation. If a customer asks whether they're talking to a bot, it still says it's an AI assistant. Leave empty to skip the introduction.</p>
<div class="form"><div class="fr"><label for="stIntro">Introduce as</label><div><div class="toolbar" style="margin:0"><input class="in" id="stIntro" maxlength="80" value="${esc(st.intro)}" placeholder="For example: Lio from the ${esc(S.settings.product || "Transtore")} Support Team" style="flex:1;min-width:220px"><button type="button" class="btn" data-act="stIntroSave">Save</button></div>
<p class="meta">The bot writes it as: "Hi there, I'm ${esc(st.intro || "…")}."</p></div></div></div></section>
<section class="sec"><div class="sec-h"><h2>Example replies from your team</h2><div class="toolbar" style="margin:0"><button type="button" class="btn sm" data-act="exImport">Import from Excel or CSV</button><button type="button" class="btn sm pri" data-act="exNew">Add</button></div></div>
<p class="hint">Paste replies your team actually sent. For each customer message the bot picks the ${n(6)} closest ones and writes in the same voice, following your writing rules above. They teach the <b>voice only</b>: facts, prices, links and promises still come from snippets and the help center, never from these. ${plural(on, "reply", "replies")} in use, up to ${n(M.maxExamples)}.</p>
${st.examples.length ? `<div class="toolbar" style="margin:0 0 12px"><input class="in search" type="search" id="exq" placeholder="Search" value="${esc(q)}" aria-label="Search example replies"><button type="button" class="btn ghost sm danger" data-act="exDelAll">Delete all</button></div>` : ""}
${
  st.examples.length
    ? list.length
      ? groups.map((g) => `${g ? `<p class="meta" style="margin:16px 0 6px;font-weight:600">${esc(g)}</p>` : ""}<div class="rows">${list.filter((e) => (e.group || "") === g).map(row).join("")}</div>`).join("")
      : `<div class="rows"><div class="empty">Nothing matches "${esc(q)}".</div></div>`
    : `<div class="rows"><div class="empty">No example replies yet. <button type="button" class="lk" data-act="exImport">Import your team's saved replies</button> (a sheet with columns like Group, Situation, Reply), or <button type="button" class="lk" data-act="exNew">add one</button>.</div></div>`
}</section>`);
  const ta = $("#stRules");
  const count = () => ($("#stRulesN").textContent = `${n(ta.value.length)} / ${n(M.maxRulesChars)} characters`);
  ta.oninput = count;
  count();
  const box = $("#exq");
  if (box) {
    box.oninput = () => {
      q = box.value;
      const pos = box.selectionStart;
      R.style();
      const b = $("#exq");
      b.focus();
      b.setSelectionRange(pos, pos);
    };
  }
};

async function saveSettings(body, msg) {
  const S = store.S;
  const r = await api("/api/style", { op: "settings", ...body }).catch((e) => {
    refresh();
    throw e;
  });
  S.style = r.style;
  refresh();
  toast(`${msg}. Try it in the Test panel.`);
}

A.stRulesSave = (el) => busy(el, () => saveSettings({ rules: $("#stRules").value }, $("#stRules").value.trim() ? "Writing rules saved" : "Writing rules removed"));
A.stRulesReset = async (el) => {
  if (!(await confirmBox("Restore the default rules?", "Your writing rules will be replaced with the default ones the bot used before this page existed.", "Restore", false))) return;
  return busy(el, () => saveSettings({ rules: null }, "Default writing rules restored"));
};
A.stIntroSave = (el) => busy(el, () => saveSettings({ intro: $("#stIntro").value }, $("#stIntro").value.trim() ? "Introduction saved" : "Introduction removed"));

/* ------------------------------ One example ------------------------------ */

function openExample(id) {
  const S = store.S;
  const e = S.style.examples.find((x) => x.id === id) || { group: "", situation: "", text: "", enabled: true };
  const groups = [...new Set(S.style.examples.map((x) => x.group).filter(Boolean))];
  modal(
    id ? "Edit example reply" : "New example reply",
    `<label class="fl" for="exg">Group <span class="muted">(optional, for example Pricing)</span></label>
<input class="in" id="exg" list="exgl" value="${esc(e.group)}" maxlength="60"><datalist id="exgl">${groups.map((g) => `<option value="${esc(g)}">`).join("")}</datalist>
<label class="fl" for="exs">Situation <span class="muted">(when this reply fits; helps the bot pick it)</span></label>
<input class="in" id="exs" value="${esc(e.situation)}" maxlength="160" placeholder="For example: Customer asks if they can use the free plan">
<label class="fl" for="ext">The reply your team sent</label>
<textarea class="in" id="ext" rows="10" maxlength="3000" placeholder="Paste the whole reply, as the customer saw it.">${esc(e.text)}</textarea>
<label class="chk"><input type="checkbox" id="exe" ${e.enabled ? "checked" : ""}> The bot learns from this reply</label>`,
    `<button class="btn" value="close">Cancel</button><button class="btn pri" type="button" data-act="exSave" data-id="${esc(id || "")}">Save reply</button>`,
    { wide: true, onOpen: (d) => $(id ? "#ext" : "#exs", d).focus() }
  );
}

A.exNew = () => openExample(null);
A.exEdit = (el) => openExample(el.dataset.id);
A.exSave = (el) =>
  busy(el, async () => {
    const S = store.S;
    const item = { id: el.dataset.id || undefined, group: $("#exg").value, situation: $("#exs").value, text: $("#ext").value, enabled: $("#exe").checked };
    const r = await api("/api/style", { op: "save", item });
    S.style = r.style;
    closeModal();
    refresh();
    toast("Example reply saved. Try it in the Test panel.");
  });
A.exToggle = (el) =>
  busy(el, async () => {
    const S = store.S;
    const e = S.style.examples.find((x) => x.id === el.dataset.id);
    const r = await api("/api/style", { op: "save", item: { ...e, enabled: !e.enabled } });
    S.style = r.style;
    refresh();
    toast(e.enabled ? "The bot no longer learns from this reply" : "The bot learns from this reply again");
  });
A.exDel = async (el) => {
  const S = store.S;
  const e = S.style.examples.find((x) => x.id === el.dataset.id);
  if (!(await confirmBox("Delete example reply?", `"${(e.situation || e.text).slice(0, 70)}" will be deleted.`))) return;
  const r = await api("/api/style", { op: "delete", id: e.id });
  S.style = r.style;
  refresh();
  toast("Example reply deleted");
};
A.exDelAll = async () => {
  const S = store.S;
  const ids = S.style.examples.map((e) => e.id);
  if (!(await confirmBox("Delete all example replies?", `All ${ids.length} example replies will be deleted. Download a backup in Settings first if you may need them again.`, "Delete all"))) return;
  const r = await api("/api/style", { op: "delete", ids });
  S.style = r.style;
  q = "";
  refresh();
  toast("All example replies deleted");
};

A.exImport = () => {
  Object.assign(imp, { sheets: [], si: 0, cols: null, items: [] });
  modal(
    "Import example replies",
    `<p class="hint" style="margin:0 0 12px">Choose an Excel (.xlsx) or CSV file with your team's saved replies: one reply per row, ideally with columns like <b>Group</b>, <b>Situation</b> and <b>Reply</b> (Nhóm, Tình huống, Nội dung also work). You pick the rows before anything is saved.</p>

<details style="margin-top:12px"><summary>Or paste cells copied from Excel or Google Sheets</summary>
<textarea class="in" id="expaste" rows="5" style="margin-top:8px" placeholder="Select the cells (with the header row), copy, paste here"></textarea>
<div class="toolbar" style="margin-top:8px"><button type="button" class="btn sm" data-act="exPasteRead">Read pasted cells</button></div></details>
<div id="exprev"></div>`,
    `<button class="btn" value="close">Cancel</button><button class="btn pri" type="button" data-act="exImportGo" id="exgo" disabled>Import</button>`,
    {
      wide: true,
      onOpen(d) {
        $("#exf", d).onchange = async (ev) => {
          const f = ev.target.files?.[0];
          if (!f) return;
          try {
            loadSheets(await readSheetFile(f));
          } catch (e) {
            toast(e?.message || "Couldn't read that file", { err: true });
          }
        };
      },
    }
  );
};

A.exPasteRead = () => {
  const rows = parseDelimited($("#expaste").value);
  if (!rows.length) return toast("Paste some cells first", { err: true });
  loadSheets([{ name: "Pasted cells", rows }]);
};

function loadSheets(sheets) {
  sheets = sheets.filter((s) => s.rows.length);
  if (!sheets.length) return toast("That file has no rows", { err: true });
  // Start on the sheet that looks most like saved replies: the most rows with real text in the reply column
  const score = (s) => toItems(s.rows, guessColumns(s.rows)).filter((x) => x.on).length;
  imp.sheets = sheets;
  imp.si = sheets.reduce((best, s, i) => (score(s) > score(sheets[best]) ? i : best), 0);
  pickSheet(imp.si);
}

function pickSheet(i) {
  imp.si = i;
  imp.cols = guessColumns(imp.sheets[i].rows);
  imp.items = toItems(imp.sheets[i].rows, imp.cols);
  drawPreview();
}

function drawPreview() {
  const sh = imp.sheets[imp.si];
  const width = Math.max(1, ...sh.rows.map((r) => r.length));
  const head = imp.cols.header >= 0 ? sh.rows[imp.cols.header] : [];
  const colOpts = (cur, none) =>
    `${none ? `<option value="-1" ${cur < 0 ? "selected" : ""}>(none)</option>` : ""}${Array.from({ length: width }, (_, c) => `<option value="${c}" ${c === cur ? "selected" : ""}>Column ${colName(c)}${head[c] ? `: ${esc(String(head[c]).slice(0, 30))}` : ""}</option>`).join("")}`;
  const picked = imp.items.filter((x) => x.on).length;
  $("#exprev").innerHTML = `<div class="form" style="margin-top:16px">
${imp.sheets.length > 1 ? `<div class="fr"><label for="exsh">Sheet</label><select class="in" id="exsh">${imp.sheets.map((s, k) => `<option value="${k}" ${k === imp.si ? "selected" : ""}>${esc(s.name)} (${plural(s.rows.length, "row")})</option>`).join("")}</select></div>` : ""}
<div class="fr"><label for="exct">Reply column</label><select class="in" id="exct">${colOpts(imp.cols.text, false)}</select></div>
<div class="fr"><label for="excs">Situation column</label><select class="in" id="excs">${colOpts(imp.cols.situation, true)}</select></div>
<div class="fr"><label for="excg">Group column</label><select class="in" id="excg">${colOpts(imp.cols.group, true)}</select></div></div>
<div class="sec-h" style="margin:16px 0 8px"><p class="meta" style="margin:0"><b>${n(picked)}</b> of ${plural(imp.items.length, "row")} ticked. Untick notes, links and replies for other apps.</p>
<div class="toolbar" style="margin:0"><button type="button" class="btn ghost sm" data-act="exTickAll" data-on="1">Tick all</button><button type="button" class="btn ghost sm" data-act="exTickAll" data-on="">Untick all</button></div></div>
<div class="rows" style="max-height:46vh;overflow:auto">${
    imp.items.length
      ? imp.items
          .map(
            (x, k) => `<label class="row" style="cursor:pointer"><input type="checkbox" data-k="${k}" ${x.on ? "checked" : ""} style="margin-top:4px;accent-color:var(--blue)">
<div class="grow"><p class="t">${esc(x.situation || x.text.split("\n")[0].slice(0, 90))}${x.group ? ` <span class="chip">${esc(x.group)}</span>` : ""}</p><p class="sub clamp">${esc(x.text)}</p></div></label>`
          )
          .join("")
      : `<div class="empty">No replies in this column. Choose another column or sheet.</div>`
  }</div>`;
  $("#exgo").disabled = !picked;
  $("#exgo").textContent = picked ? `Import ${plural(picked, "reply", "replies")}` : "Import";
  const sel = $("#exsh");
  if (sel) sel.onchange = () => pickSheet(+sel.value);
  const remap = () => {
    imp.cols = { ...imp.cols, text: +$("#exct").value, situation: +$("#excs").value, group: +$("#excg").value };
    imp.items = toItems(sh.rows, imp.cols);
    drawPreview();
  };
  ["#exct", "#excs", "#excg"].forEach((s) => ($(s).onchange = remap));
  $$("#exprev input[data-k]").forEach(
    (c) =>
      (c.onchange = () => {
        imp.items[+c.dataset.k].on = c.checked;
        const k = imp.items.filter((x) => x.on).length;
        $("#exgo").disabled = !k;
        $("#exgo").textContent = k ? `Import ${plural(k, "reply", "replies")}` : "Import";
        $("#exprev .meta b").textContent = n(k);
      })
  );
}

A.exTickAll = (el) => {
  imp.items.forEach((x) => (x.on = !!el.dataset.on));
  drawPreview();
};

A.exImportGo = (el) =>
  busy(el, async () => {
    const S = store.S;
    const items = imp.items.filter((x) => x.on).map(({ group, situation, text }) => ({ group, situation, text }));
    const r = await api("/api/style", { op: "import", items });
    const added = r.style.examples.length - S.style.examples.length;
    S.style = r.style;
    closeModal();
    refresh();
    toast(`Imported ${plural(added, "example reply", "example replies")}. Try a few questions in the Test panel.`);
  });
