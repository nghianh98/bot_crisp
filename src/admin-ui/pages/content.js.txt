// Knowledge: snippets written by the team, help center articles, website pages. The bot only answers from sources that are on.
import { $, A, R, UI, ago, api, busy, closeModal, confirmBox, esc, fail, header, host, matches, modal, n, plural, refresh, setMain, store, sw, toast } from "../core.js";

const SECTIONS = { docs: "Guides", faq: "FAQ", troubleshooting: "Troubleshooting", comparisons: "Comparisons" };
const SRC = { manual: "Written by hand", gap: "From Suggestions", import: "Bulk import", legacy: "From the old FAQ" };
const AUTO = [["weekly", "Every week"], ["daily", "Every day"], ["off", "Off, sync by hand"]];
const EVERY = { weekly: "synced every week", daily: "synced every day", off: "synced by hand" };
let syncing = false;

function inDays(t) {
  const d = Math.round((t - Date.now()) / 86400000);
  return d <= 0 ? "in the next few minutes" : d === 1 ? "tomorrow" : `in ${d} days`;
}

/** One line under the sources table: what the last sync did and when the next one runs */
function syncNote() {
  const k = store.S.kb;
  const y = k.sync || {};
  const r = y.running;
  if (r) {
    const w = Math.round((r.done / Math.max(1, r.total)) * 100);
    return `<div class="syncprog"><p class="meta">Syncing ${n(r.done)} of ${plural(r.total, "page")}…${syncing ? "" : " It finishes in the background within a few minutes."}</p><div class="track" role="progressbar" aria-valuemin="0" aria-valuemax="${r.total}" aria-valuenow="${r.done}" aria-label="Sync progress"><div class="fill" style="width:${w}%"></div></div></div>`;
  }
  const parts = [];
  const l = y.last;
  if (l) {
    let t = `Last synced ${ago(l.at)}${l.by ? ` by ${esc(l.by)}` : ""}`;
    if (l.updated != null) t += `: ${plural(l.pages, "page")}, ${n(l.updated)} updated`;
    if (l.failedCount) t += `, <span class="warn" title="${esc((l.failed || []).join("\n"))}">${plural(l.failedCount, "page")} couldn't be read</span>`;
    parts.push(`${t}.`);
  } else parts.push("Not synced yet.");
  if (y.error) parts.push(`The last automatic sync couldn't start (${esc(y.error.msg)}); it tries again every hour.`);
  else if (k.autoSync !== "off" && y.next) parts.push(`Next automatic sync ${inDays(y.next)}.`);
  else if (k.autoSync === "off") parts.push("Automatic sync is off.");
  if (k.articles.some((a) => a.src === "help")) parts.push(`<button type="button" class="lk" data-act="imgReread">Read images again</button>`);
  // Everything that is on goes to the AI with each reply: warn when that gets very large
  const size = k.articles.filter((a) => !a.off).reduce((t, a) => t + (a.chars || 0), 0);
  const big = size > 600000 ? `<p class="note warn" style="margin-top:6px">The knowledge that is on is very large (about ${n(Math.round(size / 6))} words). All of it is sent to the AI with every reply, so replies get slower and cost more. Turn off articles the bot doesn't need, or narrow Help center sections in Settings.</p>` : "";
  return `<p class="note" style="margin-top:8px">${parts.join(" ")}</p>${big}`;
}

R.content = () => {

  const web = S.kb.articles.filter((a) => a.src === "web");
  const lastSn = Math.max(0, ...sn.map((x) => x.upd || 0));
  const synced = S.kb.sync?.last?.at || S.kb.builtAt;
  const running = !!S.kb.sync?.running;
  const tab = (k, label) => `<button type="button" role="tab" aria-selected="${UI.tab === k}" data-act="ctab" data-v="${k}">${label}</button>`;

  setMain(`${header("Knowledge", "The bot only answers from the sources that are on. When a snippet and the help center disagree, the bot follows the snippet.")}
<div class="toolbar">
<button type="button" class="btn pri" data-act="snipNew">New snippet</button>
<button type="button" class="btn" data-act="snipImport">Bulk import</button>
<button type="button" class="btn" data-act="webAdd">Add website page</button>
<a class="btn" href="#suggestions">Suggestions${S.pending ? ` <span class="badge">${n(S.pending)}</span>` : ""}</a>
</div>
<section class="sec"><div class="sec-h"><h2>Content sources</h2><div class="syncbar">
<label for="autoSync">Auto sync</label><select class="in inl" id="autoSync">${AUTO.map(([v, l]) => `<option value="${v}" ${S.kb.autoSync === v ? "selected" : ""}>${l}</option>`).join("")}</select>
<button type="button" class="btn sm" data-act="syncAll" id="syncBtn" ${syncing ? "disabled" : ""}>${syncing ? "Syncing…" : running ? "Continue sync" : "Sync now"}</button></div></div>
<table class="tbl"><thead><tr><th>Source</th><th class="num">In use</th><th class="hide-sm">Updated</th></tr></thead><tbody>
${srcRow("snippets", "Snippets", "Questions and answers written by the team", sn.filter((x) => x.enabled).length, sn.length, sn.length ? ago(lastSn) : "never")}
${srcRow("help", "Help center", S.kb.helpUrl ? `${esc(host(S.kb.helpUrl))}, ${EVERY[S.kb.autoSync]}` : `Not set yet: add the link in <a href="#settings">Settings → Product</a>`, help.filter((a) => !a.off).length, help.length, ago(synced))}
${srcRow("web", "Website pages", `Pages outside the help center, like pricing; ${EVERY[S.kb.autoSync]}`, web.filter((a) => !a.off).length, web.length, web.length ? ago(synced) : "never")}
</tbody></table><div id="syncNote">${syncNote()}</div></section>
<section class="sec" id="srcList">
<div class="tabs" role="tablist">${tab("snippets", `Snippets (${sn.length})`)}${tab("help", `Help center (${help.length})`)}${tab("web", `Website (${web.length})`)}</div>
<div class="toolbar" style="margin:0 0 12px"><input class="in search" type="search" id="cq" placeholder="Search" value="${esc(UI.q)}" aria-label="Search this list"></div>
<div id="clist"></div></section>`);
  $("#autoSync").onchange = (e) => setAutoSync(e.target);
  $("#cq").oninput = (e) => {
    UI.q = e.target.value;
    renderList();
  };
  renderList();
};

function renderList() {
  const S = store.S;
  const q = UI.q;
  let html = "";
  if (UI.tab === "snippets") {
    const list = S.snippets.filter((x) => matches(q, x.q, x.a)).sort((a, b) => (b.upd || 0) - (a.upd || 0));
    html = list.length
      ? list
          .map(
            (x) => `<div class="row">${sw(x.enabled, "snipToggle", `data-id="${esc(x.id)}"`, "The bot uses this snippet")}
<div class="grow"><button type="button" class="lk t" data-act="snipEdit" data-id="${esc(x.id)}">${esc(x.q)}</button><p class="sub clamp">${esc(x.a)}</p>
<p class="meta">${esc(SRC[x.src] || "Written by hand")}, edited ${ago(x.upd)}${x.by ? ` by ${esc(x.by)}` : ""}</p></div>
<div class="acts"><button type="button" class="btn ghost sm" data-act="tryQ" data-q="${esc(x.q)}">Test</button><button type="button" class="btn ghost sm danger" data-act="snipDel" data-id="${esc(x.id)}">Delete</button></div></div>`
          )
          .join("")
      : `<div class="empty">${S.snippets.length ? "No snippets match." : "No snippets yet. A snippet is a ready answer for a question customers ask that the help center doesn't cover."}</div>`;
  } else {
    const src = UI.tab;
    const list = S.kb.articles.filter((a) => a.src === src && matches(q, a.title, a.id));
    if (!list.length) {
      html = `<div class="empty">${src === "web" ? "No website pages yet. Click Add website page to let the bot read one, for example your pricing page." : S.kb.articles.length ? "No articles match." : "The help center isn't loaded yet. Click Sync now above."}</div>`;
    } else {
      let last = null;
      for (const a of list) {
        const sec = src === "help" ? a.id.split("/")[1] : null;
        if (sec !== last && src === "help") {
          html += `<div class="grp">${esc(SECTIONS[sec] || sec)}</div>`;
          last = sec;
        }
        html += `<div class="row">${sw(!a.off, "artToggle", `data-id="${esc(a.id)}"`, "The bot uses this article")}
<div class="grow"><a class="t" href="${esc(a.url)}" target="_blank" rel="noopener">${esc(a.title)}</a><p class="meta">${esc(src === "help" ? a.id : host(a.url) + new URL(a.url).pathname)}, ${(Math.round(a.chars / 100) / 10).toFixed(1)}k characters${a.off ? ", not used by the bot" : ""}</p></div>
<div class="acts"><button type="button" class="btn ghost sm" data-act="artView" data-id="${esc(a.id)}">View text</button><button type="button" class="btn ghost sm" data-act="artSync" data-id="${esc(a.id)}">Sync page</button>${src === "web" ? `<button type="button" class="btn ghost sm danger" data-act="webRemove" data-id="${esc(a.id)}">Remove</button>` : ""}</div></div>`;
      }
    }
  }
  $("#clist").innerHTML = `<div class="rows">${html}</div>`;
}

async function setAutoSync(sel) {
  const before = store.S.kb.autoSync;
  sel.disabled = true;
  try {
    const r = await api("/api/kb", { op: "autoSync", value: sel.value });
    store.S.kb = r.kb;
    toast(sel.value === "off" ? "Automatic sync turned off. Use Sync now when the help center changes." : `Automatic sync: ${sel.selectedOptions[0].textContent.toLowerCase()}`);
    refresh();
  } catch (e) {
    sel.value = before;
    sel.disabled = false;
    fail(e);
  }
}

// An image changed but kept the same link: have the AI read every help center image again on the next sync
A.imgReread = async () => {
  if (syncing) return;
  if (!(await confirmBox("Read all images again?", "During the next sync the AI reads every image in the help center again. Use this when an image changed but kept the same link. Each image costs a little AI usage.", "Read again", false))) return;
  store.S.kb = (await api("/api/kb", { op: "rereadImages" })).kb;
  return A.syncAll();
};

// Sync all sources: start, then keep stepping while the page shows progress. Each step reads a few pages.
A.syncAll = async () => {
  if (syncing) return;
  syncing = true;
  const show = () => {
    const btn = $("#syncBtn");
    const note = $("#syncNote");
    if (btn) {
      btn.disabled = true;
      btn.textContent = "Syncing…";
    }
    if (note) note.innerHTML = syncNote();
  };
  show();
  const wait = (ms) => new Promise((r) => setTimeout(r, ms));
  let misses = 0;
  let op = "syncStart";
  try {
    while (true) {
      try {
        const r = await api("/api/kb", { op });
        store.S.kb = r.kb;
        misses = 0;
      } catch (e) {
        if (op === "syncStart" || ++misses > 4) throw e;
        await wait(1500 * misses); // a busy moment on Cloudflare: wait a little longer each time
        continue;
      }
      op = "syncStep";
      if (!store.S.kb.sync?.running) break;
      show();
    }
    const l = store.S.kb.sync?.last;
    toast(l ? `Synced ${plural(l.pages, "page")}: ${n(l.updated ?? 0)} updated${l.failedCount ? `, ${n(l.failedCount)} couldn't be read` : ""}` : "Sync finished");
  } catch (e) {
    toast(`Sync paused (${e.message}). It finishes in the background within a few minutes, or click Continue sync.`, { err: true });
  } finally {
    syncing = false;
    refresh();
  }
};

A.ctab = (el) => {
  UI.tab = el.dataset.v;
  R.content();
  document.getElementById("srcList")?.scrollIntoView({ block: "start", behavior: "smooth" });
};

A.snipToggle = (el) =>
  busy(el, async () => {
    const S = store.S;
    const x = S.snippets.find((s) => s.id === el.dataset.id);
    const r = await api("/api/snippets", { op: "save", item: { ...x, enabled: !x.enabled } });
    S.snippets = r.snippets;
    renderList();
    toast(x.enabled ? "Snippet turned off" : "Snippet turned on");
  });

A.snipDel = async (el) => {
  const S = store.S;
  const x = S.snippets.find((s) => s.id === el.dataset.id);
  if (!(await confirmBox("Delete snippet?", `"${x.q}" will be deleted and the bot stops using this answer.`))) return;
  const r = await api("/api/snippets", { op: "delete", id: x.id });
  S.snippets = r.snippets;
  refresh();
  toast("Snippet deleted");
};

A.snipImport = () =>
  modal(
    "Bulk import",
    `<p class="hint">Write each pair as a Q: line (question) and an A: line (answer), with a blank line between pairs. H: and Đ: also work.</p>
<textarea class="in" id="imp" rows="12" placeholder="Q: Do you offer free setup?
A: Yes. Send us your account email and the team sets it up within 24 working hours.

Q: Can I change my plan later?
A: …"></textarea>`,
    `<button class="btn" value="close">Cancel</button><button class="btn pri" type="button" data-act="snipImportGo">Import</button>`,
    { onOpen: (d) => d.querySelector("#imp").focus() }
  );
A.snipImportGo = (el) =>
  busy(el, async () => {
    const S = store.S;
    const before = S.snippets.length;
    const r = await api("/api/snippets", { op: "import", text: $("#imp").value });
    S.snippets = r.snippets;
    closeModal();
    UI.tab = "snippets";
    refresh();
    toast(`Imported ${r.snippets.length - before} snippets`);
  });

async function kbOp(body, done) {
  const r = await api("/api/kb", body);
  store.S.kb = r.kb;
  refresh();
  if (done) toast(done);
}

A.artToggle = (el) =>
  busy(el, () => {
    const a = store.S.kb.articles.find((x) => x.id === el.dataset.id);
    return kbOp({ op: "exclude", id: a.id, excluded: !a.off }, a.off ? "Article turned on, the bot uses it again" : "Article turned off, the bot no longer uses it");
  });
A.artSync = (el) =>
  busy(el, async () => {
    el.textContent = "Syncing…";
    await kbOp({ op: "sync", id: el.dataset.id }, "Synced the latest version of the article");
  });
A.artView = async (el) => {
  const a = store.S.kb.articles.find((x) => x.id === el.dataset.id);
  const r = await api("/api/kb", { op: "view", id: a.id });
  modal(a.title, `<p class="hint">This is the text the bot reads from this page. Anything missing here, the bot doesn't know either.</p><pre class="page">${esc(r.text || "(empty)")}</pre>`, `<a class="btn" href="${esc(a.url)}" target="_blank" rel="noopener">Open page</a><button class="btn pri" value="close">Close</button>`, { wide: true });
};
A.webAdd = () =>
  modal(
    "Add website page",
    `<p class="hint">The bot reads the text on this page and syncs it together with the help center. Good for pricing or feature pages, up to 20 pages.</p>
<label class="fl" for="wu">Page link</label><input class="in" id="wu" type="url" placeholder="https://example.com/pricing">`,
    `<button class="btn" value="close">Cancel</button><button class="btn pri" type="button" data-act="webAddGo">Add page</button>`,
    { onOpen: (d) => d.querySelector("#wu").focus() }
  );
A.webAddGo = (el) =>
  busy(el, async () => {
    el.textContent = "Reading the page…";
    try {
      const r = await api("/api/kb", { op: "addWeb", url: $("#wu").value });
      store.S.kb = r.kb;
      closeModal();
      UI.tab = "web";
      refresh();
      toast("Page added. Click View text to check what the bot can read.");
    } finally {
      el.textContent = "Add page";
    }
  });
A.webRemove = async (el) => {
  const a = store.S.kb.articles.find((x) => x.id === el.dataset.id);
  if (!(await confirmBox("Remove website page?", `The bot will stop reading ${a.url}.`, "Remove"))) return;
  await kbOp({ op: "removeWeb", id: a.id }, "Page removed");
};
