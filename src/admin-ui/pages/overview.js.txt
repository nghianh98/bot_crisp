// Overview: how the bot is doing, this month's numbers, to-dos, connections, latest replies.
import { chipsFor, esc, fmtClock, fmtDay, fmtTime, getAn, guard, header, monthLabel, n, pct, plural, R, setMain, store, tzLabel } from "../core.js";
import { fig } from "../components.js";

export function statusSentence() {
  const s = store.S.settings;
  const mode = {
    off: "The bot is off: only your team answers customers.",
    offline: "The bot answers when nobody on the team is online in Crisp.",
    always: "The bot answers every message, even when the team is online.",
  }[s.mode];
  return `${mode} ${s.businessHours ? `Team working hours: ${esc(s.businessHours)} (${tzLabel(store.S.meta.utcOffset)}).` : "Working hours aren't set."}`;
}

function todos(a) {
  const S = store.S;
  const t = [];
  for (const h of S.health.filter((x) => !x.ok)) t.push(["bad", `${h.label} isn't ready`, h.fix]);
  if (S.settings.product === "our product" || !S.settings.productDesc) t.push(["todo", "The product isn't described yet", "Add the product name and a short description so the bot knows what it supports.", "#settings", "Open settings"]);
  if (S.settings.mode === "off") t.push(["todo", "The bot is off", "Pick another mode in Bot mode on the left when you want the bot to answer customers."]);
  if (S.pending) t.push(["todo", `${plural(S.pending, "question")} the bot couldn't answer`, "Add snippets so the bot can answer them next time.", "#suggestions", "See suggestions"]);
  if (a?.feedback.down) t.push(["todo", `${plural(a.feedback.down, "customer")} rated an answer not helpful`, "Review what the bot told them.", "#conversations?down", "See conversations"]);
  if (!S.snippets.length) t.push(["todo", "No snippets yet", "Add answers to common questions the help center doesn't cover.", "#content", "Add a snippet"]);
  if (!S.settings.supportEmail) t.push(["todo", "Support email isn't set", "The bot offers this email when it hands off outside working hours.", "#settings", "Open settings"]);
  if (!S.settings.businessHours) t.push(["todo", "Working hours aren't set", "The bot uses them to tell customers when the team will reply.", "#settings", "Open settings"]);
  if (!t.length) return `<div class="row"><span class="st ok" aria-hidden="true">✓</span><div class="grow"><p class="t">Nothing to do</p><p class="sub">The bot is running well. Check Suggestions once a week.</p></div></div>`;
  return t
    .map(
      ([kind, title, sub, href, label]) =>
        `<div class="row"><span class="st ${kind}" aria-hidden="true">!</span><div class="grow"><p class="t">${esc(title)}</p><p class="sub">${esc(sub)}</p></div>${href ? `<div class="acts"><a class="btn sm" href="${href}">${esc(label)}</a></div>` : ""}</div>`
    )
    .join("");
}

R.overview = async () => {
  const S = store.S;
  setMain(`${header("Overview", statusSentence())}<p class="loading">Loading numbers…</p>`);
  const live = guard();
  const a = await getAn(S.month).catch(() => null);
  if (!live()) return;
  setMain(`${header("Overview", statusSentence())}
<section class="sec"><div class="sec-h"><h2>${monthLabel(S.month)}</h2><a href="#performance">See performance</a></div>
<div class="figs">
${fig(n(a?.customers), "conversations")}
${fig(pct(a?.involved, a?.customers), "with the bot involved", plural(a?.involved || 0, "conversation"))}
${fig(pct(a?.resolved, a?.involved), "resolved by the bot", `${n(a?.resolved)} without a handoff`)}
${fig(n(S.pending), "questions without an answer", S.pending ? `<a href="#suggestions">See suggestions</a>` : "all handled")}
</div></section>
<section class="sec"><h2>To do</h2><div class="rows">${todos(a)}</div></section>
<section class="sec"><h2>Connections</h2><div class="rows">${S.health
    .map(
      (h) =>
        `<div class="row"><span class="st ${h.ok ? "ok" : "bad"}" aria-hidden="true">${h.ok ? "✓" : "!"}</span><div class="grow"><p class="t">${esc(h.label)}</p><p class="sub">${esc(h.ok ? h.detail || "Ready" : h.fix)}</p></div></div>`
    )
    .join("")}
<div class="row"><span class="st ${S.lastReply ? "ok" : "todo"}" aria-hidden="true">${S.lastReply ? "✓" : "!"}</span><div class="grow"><p class="t">Latest bot reply</p><p class="sub">${S.lastReply ? fmtTime(S.lastReply) : "None this month yet"}${S.queued ? `. ${plural(S.queued, "conversation")} waiting for a follow-up.` : ""}</p></div></div></div></section>
<section class="sec"><div class="sec-h"><h2>Latest bot replies</h2><a href="#conversations">See all</a></div>
<div class="rows">${
    S.recent.length
      ? S.recent
          .map((x) => `<div class="row conv"><div class="when"><b>${fmtDay(x.t)}</b>${fmtClock(x.t)}</div><div class="grow"><p class="qtxt">${esc(x.q)}</p><div class="chips">${chipsFor(x)}</div></div></div>`)
          .join("")
      : `<div class="empty">The bot hasn't replied to anyone this month yet.</div>`
  }</div></section>`);
};
