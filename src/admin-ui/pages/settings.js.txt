const DAYS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"];
const TIMES = Array.from({ length: 49 }, (_, i) => `${pad(Math.floor(i / 2))}:${i % 2 ? "30" : "00"}`);
const MODES = [
  ["off", "Off", "The bot doesn't reply to anyone. Only your team answers customers."],
  ["offline", "When team is offline", "The bot only replies when nobody on the team is online in Crisp."],
  ["always", "Always on", "The bot replies to every message, even when the team is online."],
];

export function parseBH(s) {
  const m = String(s || "").trim().match(/^([A-Za-z,\- ]+)\s+(\d{1,2}):(\d{2})\s*-\s*(\d{1,2}):(\d{2})$/);
  if (!m) return null;
  const idx = (x) => DAYS.findIndex((d) => d.toLowerCase() === x.trim().slice(0, 3).toLowerCase());
  const days = new Set();
  for (const part of m[1].split(",").map((x) => x.trim()).filter(Boolean)) {
    const [a, b] = part.split("-").map(idx);
    if (a < 0 || b === -1) return null;
    if (b === undefined) days.add(a);
    else for (let d = a; ; d = (d + 1) % 7) { days.add(d); if (d === b) break; }
  }
  return { days, open: `${pad(m[2])}:${m[3]}`, close: `${pad(m[4])}:${m[5]}` };
}

export function buildBH(days, open, close) {
  const ds = [...days].sort((a, b) => a - b);
  if (!ds.length) return "";
  const parts = [];
  let start = ds[0];
  let prev = ds[0];
  for (const d of [...ds.slice(1), 99]) {
    if (d === prev + 1) { prev = d; continue; }
    parts.push(start === prev ? DAYS[start] : `${DAYS[start]}-${DAYS[prev]}`);
    start = prev = d;
  }
  return `${parts.join(",")} ${open}-${close}`;
}

A.setSave = (el) =>
  busy(el, async () => {
    const S = store.S;
    const bh = R.settings.bh();
    if (bh && $("#bo").value >= $("#bc").value) throw new Error("The end time must be after the start time");
    const settings = {
      botName: $("#bn").value,
      language: $("#lg").value,
      mode: document.querySelector('input[name="mode"]:checked')?.value,
      businessHours: bh,
      supportEmail: $("#em").value.trim(),
      model: $("#md").value.trim(),
      effort: $("#ef").value,
    };
    const r = await api("/api/settings", { settings });
    Object.assign(S, r);
    window.dispatchEvent(new Event("tb:side"));
    R.settings();
    toast("Settings saved");
  });

A.copyHook = async () => {
  try {
    await navigator.clipboard.writeText($("#wh").value);
    toast("Webhook link copied");
  } catch {
    $("#wh").select();
    toast("Press Ctrl+C (or Cmd+C) to copy");
  }
};
A.importPick = () => $("#impf").click();

async function importFile(file) {
  if (!file) return;
  let data;
  try {
    data = JSON.parse(await file.text());
  } catch {
    toast("Couldn't read that file. Choose a .json file from the Download backup button.", { err: true });
    return;
  }
  const when = data.exportedAt ? fmtTime(Date.parse(data.exportedAt)) : "an unknown date";
  if (!(await confirmBox("Restore settings?", `All current snippets, guidance, tags and settings will be replaced with the backup from ${when}. Download a backup of the current setup first.`, "Restore", true))) return;
  try {
    await api("/api/import", { data });
    toast("Restored. Reloading…");
    setTimeout(() => location.reload(), 900);
  } catch (e) {
    toast(e.message, { err: true });
  }
}

R.changelog = async () => {
  const S = store.S;
  setMain(`${header("Change log", "")}<p class="loading">Loading…</p>`);
  const live = guard();
  S.changelog = (await api("/api/state")).changelog;
  if (!live()) return;
  setMain(`${header("Change log", "Who changed what in the admin, newest first. Names come from the Your name field at sign-in.")}
<section class="sec">${
    S.changelog.length
      ? `<table class="tbl"><thead><tr><th style="width:120px">Time</th><th style="width:140px">Changed by</th><th>Change</th></tr></thead><tbody>${S.changelog
          .map((c) => `<tr><td class="num" style="text-align:left">${fmtTime(c.t)}</td><td>${esc(c.who)}</td><td>${esc(c.what)}</td></tr>`)
          .join("")}</tbody></table>`
      : `<div class="rows"><div class="empty">No changes yet.</div></div>`
  }</section>`);
};
