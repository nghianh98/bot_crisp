const DAYS = ["Mon", "Tue", "Wed", "Thu", "Fri", "Sat", "Sun"];
const TIMES = Array.from({ length: 49 }, (_, i) => `${pad(Math.floor(i / 2))}:${i % 2 ? "30" : "00"}`);
const MODES = [
  ["off", "Off", "The bot doesn't reply to anyone. Only your team answers customers."],
  ["offline", "When team is offline", "The bot only replies when nobody on the team is online in Crisp."],
  ["always", "Always on", "The bot replies to every message, even when the team is online."],
];

export function parseBH(s) {
  const m = String(s || "").trim().match(/^([A-Za-z,\- ]+)\s+(\d{1,2}):(\d{2})\s*-\s*(\d{1,2}):(\d{2})$/);
  if (!m) return null;
  const idx = (x) => DAYS.findIndex((d) => d.toLowerCase() === x.trim().slice(0, 3).toLowerCase());
  const days = new Set();
  for (const part of m[1].split(",").map((x) => x.trim()).filter(Boolean)) {
    const [a, b] = part.split("-").map(idx);
    if (a < 0 || b === -1) return null;
    if (b === undefined) days.add(a);
    else for (let d = a; ; d = (d + 1) % 7) { days.add(d); if (d === b) break; }
  }
  return { days, open: `${pad(m[2])}:${m[3]}`, close: `${pad(m[4])}:${m[5]}` };
}

export function buildBH(days, open, close) {
  const ds = [...days].sort((a, b) => a - b);
  if (!ds.length) return "";
  const parts = [];
  let start = ds[0];
  let prev = ds[0];
  for (const d of [...ds.slice(1), 99]) {
    if (d === prev + 1) { prev = d; continue; }
    parts.push(start === prev ? DAYS[start] : `${DAYS[start]}-${DAYS[prev]}`);
    start = prev = d;
  }
  return `${parts.join(",")} ${open}-${close}`;
}

/* ------------------------------ OpenAI key + model ------------------------------ */

function keyStatus(ai) {
  if (ai?.from === "admin") {
    const who = [ai.by && `by ${esc(ai.by)}`, ai.at && fmtTime(ai.at)].filter(Boolean).join(", ");
    return `✓ Key saved here: <code>${esc(ai.hint)}</code>${who ? ` (${who})` : ""}. <button type="button" class="lk" data-act="aiKeyRemove">Remove</button>`;
  }
  if (ai?.from === "cloudflare") return `✓ Using <code>OPENAI_API_KEY</code> from Cloudflare (<code>${esc(ai.hint)}</code>). A key saved here replaces it.`;
  return `<b style="color:var(--red)">No key yet.</b> The bot can't answer customers until you add one.`;
}

// S.models: null = not loaded, { loading }, { list: [{id}] }, { error }
function modelBox() {
  const S = store.S;
  const cur = $("#md")?.value?.trim() || S.settings.model;
  const M = S.models;
  if (M?.list?.length) {
    const ids = M.list.map((m) => m.id);
    const other = ids.includes(cur) ? "" : `<option value="${esc(cur)}" selected>${esc(cur)} (not available for this key)</option>`;
    return `<div class="copy" style="max-width:420px"><select class="in" id="md">${other}${ids.map((id) => `<option value="${esc(id)}" ${id === cur ? "selected" : ""}>${esc(id)}</option>`).join("")}</select><button type="button" class="btn" data-act="aiModels">Reload</button></div>
<p class="meta">${ids.length} models this key can use, newest first. The bot needs one that supports the Responses API with structured output (most GPT models do).</p>`;
  }
  const note = M?.loading ? "Loading the models your key can use…" : M?.error ? esc(M.error) : S.ai?.from ? "" : "Add a key above to choose from the models it can use.";
  return `<div class="copy" style="max-width:420px"><input class="in" id="md" value="${esc(cur)}" maxlength="60">${S.ai?.from && !M?.loading ? `<button type="button" class="btn" data-act="aiModels">Load models</button>` : ""}</div>${note ? `<p class="meta">${note}</p>` : ""}`;
}

function paintModels() {
  const box = $("#mdBox");
  if (box) box.innerHTML = modelBox();
}

async function loadModels() {
  const S = store.S;
  S.models = { loading: true };
  paintModels();
  try {
    S.models = { list: (await api("/api/openai", { op: "models" })).models };
  } catch (e) {
    S.models = { error: `Couldn't load models: ${e.message}` };
  }
  paintModels();
}

A.aiModels = (el) => busy(el, loadModels);

A.aiKeySave = (el) =>
  busy(el, async () => {
    const key = $("#ak").value.trim();
    if (!key) throw new Error("Paste the key first");
    const r = await api("/api/openai", { op: "saveKey", key });
    const S = store.S;
    Object.assign(S, await api("/api/state"));
    S.models = r.models ? { list: r.models } : { error: r.warning };
    R.settings();
    toast(r.warning || "Key saved and checked with OpenAI");
  });

A.aiKeyRemove = async () => {
  const S = store.S;
  const text = S.ai?.cloudflare ? "The bot goes back to the OPENAI_API_KEY set in Cloudflare." : "No other key is set, so the bot stops answering customers until you add a key.";
  if (!(await confirmBox("Remove the saved key?", text, "Remove", true))) return;
  try {
    await api("/api/openai", { op: "removeKey" });
    Object.assign(S, await api("/api/state"));
    S.models = null;
    R.settings();
    toast("Key removed");
  } catch (e) {
    toast(e.message, { err: true });
  }
};

A.setSave = (el) =>
  busy(el, async () => {
    const S = store.S;
    const bh = R.settings.bh();
    if (bh && $("#bo").value >= $("#bc").value) throw new Error("The end time must be after the start time");
    const settings = {
      botName: $("#bn").value,
      language: $("#lg").value,
      mode: document.querySelector('input[name="mode"]:checked')?.value,
      businessHours: bh,
      supportEmail: $("#em").value.trim(),
      product: $("#pn").value.trim(),
      productDesc: $("#pd").value.trim(),
      helpUrl: $("#hu").value.trim().replace(/\/+$/, ""),
      helpSections: $("#hs").value.trim(),
      model: $("#md").value.trim(),
      effort: $("#ef").value,
    };
    const r = await api("/api/settings", { settings });
    Object.assign(S, r);
    window.dispatchEvent(new Event("tb:side"));
    R.settings();
    toast(r.helpSync ? "Settings saved. The help center is syncing again: see progress on the Knowledge page." : "Settings saved");
  });

A.copyHook = async () => {
  try {
    await navigator.clipboard.writeText($("#wh").value);
    toast("Webhook link copied");
  } catch {
    $("#wh").select();
    toast("Press Ctrl+C (or Cmd+C) to copy");
  }
};
A.importPick = () => $("#impf").click();

async function importFile(file) {
  if (!file) return;
  let data;
  try {
    data = JSON.parse(await file.text());
  } catch {
    toast("Couldn't read that file. Choose a .json file from the Download backup button.", { err: true });
    return;
  }
  const when = data.exportedAt ? fmtTime(Date.parse(data.exportedAt)) : "an unknown date";
  if (!(await confirmBox("Restore settings?", `All current snippets, guidance, tags and settings will be replaced with the backup from ${when}. Download a backup of the current setup first.`, "Restore", true))) return;
  try {
    await api("/api/import", { data });
    toast("Restored. Reloading…");
    setTimeout(() => location.reload(), 900);
  } catch (e) {
    toast(e.message, { err: true });
  }
}

          .join("")}</tbody></table>`
      : `<div class="rows"><div class="empty">No changes yet.</div></div>`
  }</section>`);
};
