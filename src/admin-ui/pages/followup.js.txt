// Feedback & follow-up: Helpful / Not helpful buttons after an answer, nudges when the customer goes quiet, closing the chat.
import { $, A, R, api, busy, esc, header, setMain, store, sw, toast } from "../core.js";

const MIN = [5, 10, 15, 30, 60, 120, 240];
const HOURS = [1, 2, 4, 8, 12, 24, 48, 72];
const opt = (list, v, unit) => list.map((x) => `<option value="${x}" ${x === v ? "selected" : ""}>${x} ${unit}</option>`).join("");
const hours = (h) => `${h} ${h === 1 ? "hour" : "hours"}`;

function flow(F) {
  const step = (on, em, b, s) => `<div class="fstep${on ? "" : " off"}"><em>${em}</em><b>${b}</b><span>${s}</span></div>`;
  return `<div class="flow" aria-label="What happens after the bot replies, in order">
${step(true, "Step 1", "Bot replies", "with a link to the guide")}
${step(F.feedback, "Right after", "Asks if it helped", `2 buttons: ${esc(F.yesLabel)} / ${esc(F.noLabel)}`)}
${step(F.nudge, `Quiet for ${F.nudgeMin} min`, "Nudges once", "only if the customer hasn't replied")}
${step(F.close, `Quiet for ${hours(F.closeHours)}`, "Closes the chat", F.autoResolve ? "and resolves it in Crisp" : "logged as no reply from the customer")}</div>`;
}

R.followup = () => {
  const S = store.S;
  const F = S.followup;
  const txt = (id, v, max, rows = 2) => `<textarea class="in" id="${id}" rows="${rows}" maxlength="${max}" style="min-height:0">${esc(v)}</textarea>`;
  setMain(`${header("Feedback & follow-up", "After the bot replies, ask the customer whether the answer helped, and nudge them when they go quiet. No nudges once the bot has handed off or someone on the team has joined.")}
<section class="sec">${flow(F)}</section>
<section class="sec"><h2>Ask for feedback after an answer</h2><div class="form">
<div class="fr"><div class="lb">Send 2 buttons<small>Not sent when the customer is angry or the bot had no answer</small></div><div>${sw(F.feedback, "fuFlip", `data-k="feedback"`, "Send feedback buttons")}</div></div>
<div class="fr"><label for="fq">Question</label>${txt("fq", F.feedbackText, 120, 1)}</div>
<div class="fr"><div class="lb">Button text</div><div class="times"><input class="in" id="fy" value="${esc(F.yesLabel)}" maxlength="40" aria-label="Helpful button"><input class="in" id="fn" value="${esc(F.noLabel)}" maxlength="40" aria-label="Not helpful button"></div></div>
<div class="fr"><label for="ft">Customer clicks helpful<small>The bot thanks them with this message. If they click not helpful, the bot asks what's still not working.</small></label>${txt("ft", F.thanksText, 300)}</div>
</div></section>
<section class="sec"><h2>Nudge when the customer goes quiet</h2><div class="form">
<div class="fr"><div class="lb">After an answer<small>Only nudges once</small></div><div>${sw(F.nudge, "fuFlip", `data-k="nudge"`, "Nudge after an answer")}
<div class="times">Nudge after <select class="in" id="nm" aria-label="Minutes">${opt(MIN, F.nudgeMin, "min")}</select></div>
<label class="fl" for="nt">Nudge message</label>${txt("nt", F.nudgeText, 300)}</div></div>
<div class="fr"><div class="lb">While the bot waits for details<small>For example, it asked for the store URL and the customer hasn't sent it</small></div><div>${sw(F.infoNudge, "fuFlip", `data-k="infoNudge"`, "Nudge while waiting for details")}
<div class="times">Nudge after <select class="in" id="im" aria-label="Minutes">${opt(MIN, F.infoNudgeMin, "min")}</select></div>
<label class="fl" for="it">Nudge message</label>${txt("it", F.infoNudgeText, 300)}</div></div>
</div></section>
<section class="sec"><h2>Close the chat</h2><div class="form">
<div class="fr"><div class="lb">Close when the customer goes quiet<small>Logged in reports as no reply from the customer</small></div><div>${sw(F.close, "fuFlip", `data-k="close"`, "Close the chat")}
<div class="times">After <select class="in" id="ch" aria-label="Hours">${opt(HOURS, F.closeHours, "h")}</select> from the bot's last reply</div></div></div>
<div class="fr"><div class="lb">Resolve in Crisp automatically<small>Needs a Production token with write access to website:conversation:states</small></div><div>${sw(F.autoResolve, "fuFlip", `data-k="autoResolve"`, "Resolve automatically")}
<p class="meta">If the token only has read access, the bot skips this step and everything else still runs.</p></div></div>
</div></section>
<div class="savebar"><p class="meta">Changes reach real customers within about a minute.</p><button type="button" class="btn pri" data-act="fuSave">Save</button></div>`);
};

A.fuFlip = (el) => el.setAttribute("aria-checked", String(el.getAttribute("aria-checked") !== "true"));

A.fuSave = (el) =>
  busy(el, async () => {
    const S = store.S;
    const on = (k) => document.querySelector(`.sw[data-k="${k}"]`).getAttribute("aria-checked") === "true";
    const followup = {
      feedback: on("feedback"),
      feedbackText: $("#fq").value,
      yesLabel: $("#fy").value,
      noLabel: $("#fn").value,
      thanksText: $("#ft").value,
      nudge: on("nudge"),
      nudgeMin: +$("#nm").value,
      nudgeText: $("#nt").value,
      infoNudge: on("infoNudge"),
      infoNudgeMin: +$("#im").value,
      infoNudgeText: $("#it").value,
      close: on("close"),
      closeHours: +$("#ch").value,
      autoResolve: on("autoResolve"),
    };
    const r = await api("/api/settings", { followup });
    Object.assign(S, r);
    R.followup();
    toast("Feedback and follow-up settings saved");
  });
