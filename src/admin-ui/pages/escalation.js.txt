// Escalation: when to hand off to the team, what to ask first, how to rate priority, what the bot does while the customer waits.
// Everything here is the team's own wording: the code only supplies neutral starting text.
import { $, $$, A, R, api, busy, esc, header, setMain, store, sw, toast } from "../core.js";

const RESET_HOURS = [0, 1, 2, 4, 8, 12, 24, 48];

// One editable line of a list (details to ask for, the team's own hand-off reasons)
const itemRow = (list, x, placeholder) => `<div class="row li" data-list="${list}">${sw(x.enabled !== false, "escItemFlip", "", "Use this line")}
<div class="grow"><input class="in" data-id="${esc(x.id || "")}" value="${esc(x.text || "")}" maxlength="300" placeholder="${esc(placeholder)}" aria-label="${esc(placeholder)}"></div>
<div class="acts"><button type="button" class="btn ghost sm danger" data-act="escItemDel">Delete</button></div></div>`;

const PLACEHOLDER = {
  fields: "For example: the account email, the order number, the page where it happens",
  custom: "For example: the customer wants to change the account owner",
};

const restore = (k) => `<button type="button" class="lk" data-act="escDefault" data-k="${k}">Restore default</button>`;

R.escalation = () => {
  const S = store.S;
  const E = S.escalation;
  const M = S.meta;
  const D = M.defaults;
  const sg = S.tags.segments;
  const hours = RESET_HOURS.includes(E.resetHours) ? RESET_HOURS : [...RESET_HOURS, E.resetHours].sort((a, b) => a - b);
  setMain(`${header("Escalation", "Choose when the bot hands a conversation to the team, what it asks first, and how it behaves around a handoff. All wording on this page is yours to edit. Click Save to apply.")}
<section class="sec"><h2>The bot hands off when</h2><div class="rows">${Object.entries(M.triggers)
    .map(([k, t]) => `<div class="row">${sw(E.triggers[k], "escFlip", `data-k="${k}"`, t.label)}<div class="grow"><p class="t">${esc(t.label)}</p><p class="sub">${esc(t.hint)}</p></div></div>`)
    .join("")}</div>
<div class="sec-h" style="margin-top:18px"><h3 class="h3">Your own reasons</h3><button type="button" class="btn sm" data-act="escItemAdd" data-list="custom">Add a reason</button></div>
<p class="hint">Situations in your product that always need a person. Write each one as a short phrase.</p>
<div class="rows" id="list-custom">${E.custom.map((x) => itemRow("custom", x, PLACEHOLDER.custom)).join("") || `<div class="empty">No reasons of your own yet.</div>`}</div>
<p class="note">With everything in this section off, the bot never hands off and always answers itself. The team can still reply in any conversation at any time.</p></section>

<section class="sec"><h2>Ask for details before handing off</h2><div class="rows"><div class="row">${sw(E.askInfo, "escFlip", `data-k="askInfo"`, "Ask for details before handing off")}
<div class="grow"><p class="t">Ask the customer once before handing off</p><p class="sub">The bot asks for any missing details in one message, then hands off. It never asks for something your app already sends to Crisp. If the customer asks for a person, is angry or has a payment dispute, it hands off right away.</p></div></div></div>
<div class="sec-h" style="margin-top:18px"><h3 class="h3">Details to ask for</h3><button type="button" class="btn sm" data-act="escItemAdd" data-list="fields">Add a detail</button></div>
<p class="hint">What your team needs to start working on a case. Write each one the way the bot should ask for it. Never list passwords or card numbers.</p>
<div class="rows" id="list-fields">${E.fields.map((x) => itemRow("fields", x, PLACEHOLDER.fields)).join("") || `<div class="empty">No details listed: the bot hands off without asking.</div>`}</div></section>

<section class="sec"><div class="sec-h"><h2>Priority and urgency</h2>${restore("priority")}</div>
<p class="hint">The bot rates every conversation P1–P4 with your definitions. P1 always goes to a person right away; P1 and P2 get the <b>${esc(sg.p1)}</b> and <b>${esc(sg.p2)}</b> segments in Crisp.</p>
<div class="form">${Object.entries(M.priorityLevels)
    .map(([k, name]) => `<div class="fr"><label for="pr_${k}">${esc(name)}</label><textarea class="in" id="pr_${k}" rows="2" maxlength="400">${esc(E.priority[k])}</textarea></div>`)
    .join("")}
<div class="fr"><label for="urg">Urgent<small>Adds the <b>${esc(sg.urgent)}</b> segment. Finish the sentence "urgent when…"</small></label><textarea class="in" id="urg" rows="2" maxlength="400">${esc(E.urgent)}</textarea></div></div></section>

<section class="sec"><div class="sec-h"><h2>Upset customers</h2>${restore("upset")}</div>
<p class="hint">How the bot talks to a customer who is angry or upset, one rule per line. Leave empty for no special rules.</p>
<textarea class="in" id="upset" rows="7" maxlength="${M.maxRuleChars}" spellcheck="false">${esc(E.upset ?? D.upset)}</textarea></section>

<section class="sec"><h2>Note for the team</h2><div class="rows"><div class="row">${sw(E.note, "escFlip", `data-k="note"`, "Leave a note for the team")}
<div class="grow"><p class="t">Leave a private note when handing off</p><p class="sub">The bot adds a short brief to the conversation in Crisp: why it needs a person, what the customer needs, the details they gave, what's still missing and what to do first. Only the team sees it, never the customer. Try it in Test the bot.</p></div></div></div>
<div class="form" style="margin-top:12px"><div class="fr"><label for="nl">Written in</label><input class="in" id="nl" maxlength="30" value="${esc(E.noteLang)}" placeholder="English"></div>
<div class="fr"><label for="nx">What the note should include<small>Optional. Anything your team always wants to see.</small></label><textarea class="in" id="nx" rows="3" maxlength="600" placeholder="For example: always include the customer's plan and the page where the problem happens.">${esc(E.noteExtra)}</textarea></div></div></section>

<section class="sec"><h2>While the customer waits for the team</h2><div class="rows">
<div class="row"><div class="grow"><p class="t">The customer keeps messaging</p><p class="sub">Until someone on the team replies, the bot keeps answering, at most one message every
<select class="in inl" id="cd" aria-label="Minutes between bot messages after a handoff">${[1, 2, 5, 10, 30].map((v) => `<option value="${v}" ${E.cooldownMin === v ? "selected" : ""}>${v} min</option>`).join("")}</select>. Messages sent sooner are answered when that time is up (within about a minute), never skipped.</p></div></div>
<div class="row"><div class="grow"><p class="t">The customer comes back later</p><p class="sub">A customer who writes again after a quiet period of
<select class="in inl" id="rh" aria-label="Quiet hours before a new chat">${hours.map((v) => `<option value="${v}" ${E.resetHours === v ? "selected" : ""}>${v ? `${v} h` : "never"}</option>`).join("")}</select> starts a new chat: the bot answers normally and only mentions the earlier case if they ask. The case stays in the team's inbox.</p></div></div>
<div class="row"><div class="grow"><p class="t">Marked in Crisp</p><p class="sub">The conversation gets the <b>${esc(sg.handoff)}</b> segment. When someone on the team replies, the bot goes quiet and adds <b>${esc(sg.paused)}</b>. Resolving the conversation turns the bot back on. Rename segments on the Tags page.</p></div></div>
</div>
<div class="sec-h" style="margin-top:18px"><h3 class="h3">How the bot replies while they wait</h3>${restore("waiting")}</div>
<p class="hint">Rules for the bot's replies after a handoff, until the team answers. Leave empty for no special rules.</p>
<textarea class="in" id="waiting" rows="7" maxlength="${M.maxRuleChars}" spellcheck="false">${esc(E.waiting ?? D.waiting)}</textarea></section>
<div class="savebar"><p class="meta">Changes reach real customers within about a minute.</p><button type="button" class="btn pri" data-act="escSave">Save</button></div>`);
};

A.escFlip = (el) => el.setAttribute("aria-checked", String(el.getAttribute("aria-checked") !== "true"));
A.escItemFlip = A.escFlip;

A.escItemAdd = (el) => {
  const S = store.S;
  const list = el.dataset.list;
  const box = $(`#list-${list}`);
  const max = list === "fields" ? S.meta.maxFields : S.meta.maxCustomReasons;
  if ($$(".li", box).length >= max) return toast(`Up to ${max} lines. Delete one first.`, { err: true });
  $(".empty", box)?.remove();
  box.insertAdjacentHTML("beforeend", itemRow(list, { enabled: true }, PLACEHOLDER[list]));
  $$(".li input", box).at(-1).focus();
};
A.escItemDel = (el) => el.closest(".li").remove();

A.escDefault = (el) => {
  const D = store.S.meta.defaults;
  const k = el.dataset.k;
  if (k === "priority") {
    for (const p of Object.keys(D.priority)) $(`#pr_${p}`).value = D.priority[p];
    $("#urg").value = D.urgent;
  } else $(`#${k}`).value = D[k];
  toast("Default text filled in. Click Save to apply.");
};

const readList = (list) =>
  $$(`.li[data-list="${list}"]`).map((row) => {
    const input = $("input", row);
    return { id: input.dataset.id || undefined, text: input.value, enabled: $(".sw", row).getAttribute("aria-checked") === "true" };
  });

A.escSave = (el) =>
  busy(el, async () => {
    const S = store.S;
    const val = (k) => document.querySelector(`.sw[data-k="${k}"]`).getAttribute("aria-checked") === "true";
    const escalation = {
      triggers: Object.fromEntries(Object.keys(S.meta.triggers).map((k) => [k, val(k)])),
      custom: readList("custom"),
      askInfo: val("askInfo"),
      fields: readList("fields"),
      cooldownMin: +$("#cd").value,
      resetHours: +$("#rh").value,
      note: val("note"),
      noteLang: $("#nl").value,
      noteExtra: $("#nx").value,
      priority: Object.fromEntries(Object.keys(S.meta.priorityLevels).map((k) => [k, $(`#pr_${k}`).value])),
      urgent: $("#urg").value,
      upset: $("#upset").value,
      waiting: $("#waiting").value,
    };
    const r = await api("/api/settings", { escalation });
    Object.assign(S, r);
    R.escalation();
    toast("Escalation settings saved");
  });
